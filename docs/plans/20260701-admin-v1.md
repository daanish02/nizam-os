# Admin v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deploy a code-only health monitor that checks all tracked services every 2 minutes, auto-restarts critical ones (cap: 2/hour), and posts Discord alerts only on state transitions (up→down or down→up).

**Architecture:** `scripts/health-monitor.py` is a standalone Python script run via systemd oneshot timer every 2 minutes. It reads `inventory/tracked-services.txt` to know which services to check and their tier (critical/watch). State is persisted in Redis between runs so alerts only fire once per transition, not every 2 minutes. The script is code-only — no LLM in the alert path. Nazim's Hermes cron runs a separate daily health summary (human-readable, LLM-generated).

**Tech Stack:** Python 3.12, redis-py, httpx, systemd oneshot timer, subprocess (systemctl).

## Global Constraints

- **Never run `git commit`** — user commits manually.
- Health monitor is a **system-level** unit (not user). It runs even when Hermes is dead and survives logout.
- RESTART_CAP = 2 — at most 2 restart attempts per service per hour. Tracked in Redis with 3600s TTL.
- Alert fires ONCE per state transition (up→down, down→up). Subsequent cycles while service is down do NOT re-alert unless another restart attempt is made or the cap message fires.
- The script filename uses a hyphen (`health-monitor.py`) to match existing naming convention. Tests use `importlib.util` to load it since hyphens are not valid Python identifiers.
- `scripts/health-monitor.py` runs under `vazir` user, not root. System-level service restarts go through `sudo systemctl restart` (covered by `/etc/sudoers.d/nazim-hermes` from Plan 1 Task 16).
- User-level services (hermes-gateway-*.service, hermes-profile-watcher.service) require `systemctl --user` with `DBUS_SESSION_BUS_ADDRESS` set from `/run/user/<uid>/bus`.
- Prerequisite: Plan 1 (immediate fixes) complete. Prerequisite: Plan 2 (curator v1) complete (so `knowledge-service.service` exists in inventory).

Create `scripts/health-monitor.py`:

```python
#!/usr/bin/env python3
"""Code-only system health monitor for nizam-os.
Checks all tracked services, auto-restarts critical ones (cap: 2/hour),
posts Discord alerts on state transitions.
Runs every 2 minutes via health-monitor.timer.
"""

import os
import subprocess
from datetime import datetime, timezone
from pathlib import Path
from typing import NamedTuple

import httpx
import redis as redis_lib

INVENTORY_PATH = Path(__file__).parent.parent / "inventory" / "tracked-services.txt"
RESTART_CAP = 2


class ServiceEntry(NamedTuple):
    name: str
    svc_type: str  # "system" | "user"
    tier: str      # "critical" | "watch"


def parse_inventory(path: Path) -> list[ServiceEntry]:
    """Parse tracked-services.txt into ServiceEntry list.

    Format: service_name | type | expected_status  [# critical|watch]
    Skips comment lines, blank lines, and lines containing '| inactive'.
    """
    entries = []
    for line in path.read_text().splitlines():
        stripped = line.strip()
        if not stripped or stripped.startswith("#"):
            continue
        if "| inactive" in stripped:
            continue

        tier = "critical" if "# critical" in stripped else "watch"
        clean = stripped.split("#")[0].strip()
        parts = [p.strip() for p in clean.split("|")]
        if len(parts) < 2:
            continue

        entries.append(ServiceEntry(name=parts[0], svc_type=parts[1], tier=tier))
    return entries


def get_dbus_address(uid: int) -> str:
    return f"unix:path=/run/user/{uid}/bus"


def check_active(entry: ServiceEntry, dbus_addr: str) -> bool:
    """Return True if the service reports 'active'."""
    env = os.environ.copy()
    if entry.svc_type == "user":
        env["DBUS_SESSION_BUS_ADDRESS"] = dbus_addr
        cmd = ["systemctl", "--user", "is-active", entry.name]
    else:
        cmd = ["systemctl", "is-active", entry.name]

    result = subprocess.run(cmd, capture_output=True, text=True, env=env)
    return result.stdout.strip() == "active"


def restart_service(entry: ServiceEntry, dbus_addr: str) -> bool:
    """Restart a service. Returns True if active after restart."""
    env = os.environ.copy()
    if entry.svc_type == "user":
        env["DBUS_SESSION_BUS_ADDRESS"] = dbus_addr
        subprocess.run(["systemctl", "--user", "restart", entry.name], env=env)
    else:
        subprocess.run(["sudo", "systemctl", "restart", entry.name])
    return check_active(entry, dbus_addr)


def build_down_embed(name: str, tier: str, action: str) -> dict:
    return {
        "embeds": [{
            "title": "SERVICE DOWN",
            "color": 15158332,
            "fields": [
                {"name": "Service", "value": name, "inline": True},
                {"name": "Tier", "value": tier, "inline": True},
                {"name": "Action", "value": action, "inline": False},
            ],
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }]
    }


def build_recovery_embed(name: str, down_checks: int) -> dict:
    return {
        "embeds": [{
            "title": "SERVICE RECOVERED",
            "color": 3066993,
            "fields": [
                {"name": "Service", "value": name, "inline": True},
                {"name": "Was down", "value": f"{down_checks} checks", "inline": True},
            ],
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }]
    }


def post_embed(webhook_url: str, embed: dict) -> None:
    if not webhook_url:
        return
    try:
        httpx.post(webhook_url, json=embed, timeout=10)
    except Exception:
        pass  # Cannot alert about alert failures


def main() -> None:
    redis_url = os.environ.get("REDIS_URL", "redis://localhost:6379/0")
    webhook = os.environ.get("DISCORD_ADMIN_WEBHOOK", "")
    r = redis_lib.from_url(redis_url, decode_responses=True)
    dbus_addr = get_dbus_address(os.getuid())
    services = parse_inventory(INVENTORY_PATH)

    for entry in services:
        is_up = check_active(entry, dbus_addr)
        state_key = f"nizam:health:state:{entry.name}"
        restart_key = f"nizam:health:restarts:{entry.name}"
        down_count_key = f"nizam:health:down_count:{entry.name}"
        cap_key = f"nizam:health:cap_posted:{entry.name}"
        prev = r.get(state_key) or "up"

        if is_up:
            if prev == "down":
                down_checks = int(r.get(down_count_key) or 0)
                r.set(state_key, "up")
                r.delete(restart_key)
                r.delete(down_count_key)
                r.delete(cap_key)
                post_embed(webhook, build_recovery_embed(entry.name, down_checks))
        else:
            r.incr(down_count_key)
            if prev == "up":
                r.set(state_key, "down")

            if entry.tier == "critical":
                count = int(r.get(restart_key) or 0)
                if count < RESTART_CAP:
                    success = restart_service(entry, dbus_addr)
                    r.incr(restart_key)
                    r.expire(restart_key, 3600)
                    result_str = "active" if success else "still down"
                    action = f"Restarted ({count + 1}/{RESTART_CAP}) — {result_str}"
                    post_embed(webhook, build_down_embed(entry.name, entry.tier, action))
                elif not r.exists(cap_key):
                    r.set(cap_key, "1", ex=3600)
                    post_embed(webhook, build_down_embed(
                        entry.name, entry.tier,
                        "Restart cap reached — manual intervention required",
                    ))
            elif prev == "up":
                post_embed(webhook, build_down_embed(
                    entry.name, entry.tier, "No action (watch tier)"
                ))


if __name__ == "__main__":
    main()
```
